# Import libraries
from cpmpy import *
import json


def among(m,x,v):
  """
  among(m,x,v)

  Requires exactly m variables in x to take one of the values in v.
  """
  return [m == sum([x[i] == j for i in range(len(x)) for j in v])]


# Decision variables
x = intvar(0, 7, shape=n, name="x")

# Model
model = Model([
    among(m, x, v)
])

# Solve the model
model.solve()

# Print the solution
solution = {"x": x.value().tolist()}
print(json.dumps(solution))
# End of CPMPy script
